NOKIA SR LINUX READ-ONLY AGENT REFERENCE | SR Linux 24.10
Source: learn.srlinux.dev + documentation.nokia.com/srlinux

--- TRANSPORT ---
GET  = JSON-RPC method:"get" datastore:"state" path:"<YANG-path>" — structured JSON, use for specific state leaves/containers
CLI  = JSON-RPC method:"cli" commands:["<show cmd>"] — use for CLI plugins that aggregate multiple YANG paths (no single path equivalent)
RULE: YANG path listed → prefer GET. No path listed → use CLI.

--- S1: PLATFORM ---
CLI  show platform chassis                  # no 'detail' option exists
CLI  show platform environment
CLI  show platform control * detail         # cpu / memory / uptime here, not show system cpu/memory
CLI  show platform linecard * detail
# NOTE: no 'show platform fabric' on 7220 IXR fixed chassis
GET  /platform/fan-tray[id=*]              # speed, oper-state
GET  /platform/power-supply[id=*]          # oper-state, capacity, used-capacity
GET  /platform/control[slot=A]/cpu[index=*]
GET  /platform/control[slot=A]/cpu[index=all]/total/instant  # single instant CPU% reading
GET  /platform/control[slot=A]/memory
GET  /platform/control[slot=A]/memory/utilization  # utilization leaf (NOT 'utilized')
GET  /system/information                   # version, hostname, chassis-type, last-booted
CLI  show version                          # brief version string (less detail than GET /system/information)

--- S2: INTERFACES ---
# Names: ethernet-1/1..N, ethernet-2/1..N, mgmt0, lo0/lo1, system0
CLI  show interface                        # all UP interfaces + subinterfaces
CLI  show interface brief                  # all interfaces tabular (admin/oper/speed)
CLI  show interface all                    # all interfaces including DOWN
CLI  show interface ethernet-1/1           # single interface + reason-if-down
CLI  show interface ethernet-1/1 detail    # includes counters — use this instead of 'statistics' subcommand
GET  /interface[name=ethernet-1/1]/oper-state
GET  /interface[name=ethernet-1/1]/oper-down-reason  # reason string when oper-state=down
GET  /interface[name=ethernet-1/1]/admin-state
GET  /interface[name=*]/oper-state         # all interfaces bulk
GET  /interface[name=ethernet-1/1]         # full: oper-state, admin-state, mtu, ifindex, last-change, ethernet/hw-mac-address, ethernet/port-speed, subinterface[index=0]/ipv4/address
GET  /interface[name=ethernet-1/1]/statistics  # in-octets, out-octets, in-errors, out-errors, in-discards, out-discards, in-unicast-pkts, out-unicast-pkts
GET  /interface[name=ethernet-1/1]/statistics/in-octets   # specific leaf
GET  /interface[name=ethernet-1/1]/statistics/out-octets  # specific leaf
GET  /interface[name=ethernet-1/1]/subinterface[index=0]  # oper-state, ipv4/address, vlan/encap
GET  /interface[name=ethernet-1/1]/subinterface[index=0]/ipv4/address  # explicit IP address leaf
GET  /interface[name=ethernet-1/1]/transceiver  # oper-state, form-factor, vendor, serial-number, channel[index=1]/output-power, channel[index=1]/input-power

--- S3: NETWORK INSTANCES ---
CLI  show network-instance summary         # all NIs: type, admin/oper-state, router-id, interface count (bare 'show network-instance' only prints usage help — always use 'summary')
GET  /network-instance[name=default]       # type, admin-state, oper-state, router-id, description
GET  /network-instance[name=default]/oper-state  # specific oper-state leaf
GET  /network-instance[name=*]
GET  /network-instance[name=*]/oper-state  # bulk oper-state all NIs
GET  /network-instance[name=default]/interface[name=*]   # interfaces assigned: name, oper-state, index
# ARP/ND live under the interface subinterface (ipv4/arp), NOT under network-instance.
GET  /interface[name=*]/subinterface[index=*]/ipv4/arp/neighbor[ipv4-address=*]   # all ARP entries: link-layer-address (MAC), origin (dynamic/static), expiry-time
GET  /interface[name=ethernet-1/1]/subinterface[index=0]/ipv4/arp/neighbor[ipv4-address=192.168.1.1]  # single entry
GET  /interface[name=*]/subinterface[index=*]/ipv6/neighbor-discovery/neighbor[ipv6-address=*]   # IPv6 ND: link-layer-address, origin, is-router
CLI  show arpnd arp-entries                 # all ARP entries (bare command — no network-instance arg)
CLI  show network-instance <mac-vrf> bridge-table mac-table all     # MAC table, mac-vrf only
CLI  show network-instance <mac-vrf> bridge-table mac-table summary
CLI  show network-instance default route-table                       # IPv4 FIB+RIB, no single YANG path
CLI  show network-instance default route-table all
CLI  show network-instance default route-table ipv4-unicast summary
GET  /network-instance[name=default]/route-table/ipv4-unicast/route[ipv4-prefix=10.0.0.0/24][route-type=bgp][route-owner=bgp_mgr]  # NOTE: route-type+route-owner are key — if unknown use CLI
CLI  show network-instance default route-table ipv4-unicast prefix 10.0.0.0/24  # alt to GET above
CLI  show network-instance default route-table ipv6-unicast summary

--- S4: BGP ---
# All BGP scoped per network-instance
CLI  show network-instance default protocols bgp summary             # peer count, paths, AFI/SAFI overview — CLI plugin
CLI  show network-instance default protocols bgp neighbor            # all peers with state column
GET  /network-instance[name=default]/protocols/bgp/neighbor[peer-address=10.0.0.1]/session-state  # Established/Active/Connect/Idle/OpenSent/OpenConfirm
GET  /network-instance[name=default]/protocols/bgp/neighbor[peer-address=10.0.0.1]/peer-as
GET  /network-instance[name=default]/protocols/bgp/neighbor[peer-address=10.0.0.1]/last-established
GET  /network-instance[name=default]/protocols/bgp/neighbor[peer-address=*]/session-state  # bulk all peers
GET  /network-instance[name=default]/protocols/bgp/neighbor[peer-address=*]
CLI  show network-instance default protocols bgp neighbor 10.0.0.1 detail  # capabilities, timers, AFI/SAFI negotiated
# NOTE: BGP route subcommands use 'ipv4' / 'ipv6' (NOT 'ipv4-unicast' / 'ipv6-unicast')
CLI  show network-instance default protocols bgp neighbor 10.0.0.1 received-routes ipv4
CLI  show network-instance default protocols bgp neighbor 10.0.0.1 advertised-routes ipv4
CLI  show network-instance default protocols bgp routes ipv4 summary
CLI  show network-instance default protocols bgp routes ipv6 summary
CLI  show network-instance default protocols bgp routes ipv4 prefix 10.0.0.0/24 detail



--- S5: ACL ---
# NOTE: filters live under /acl/acl-filter (NOT ipv4-filter/ipv6-filter); distinguish v4/v6 with the 'type' key
GET  /acl/acl-filter[name=*]                                              # all ACL filters (ipv4 and ipv6)
GET  /acl/acl-filter[name=FILTER-IN][type=ipv4]/entry[sequence-id=*]      # sequence-id, match(protocol/src/dst/port), action(accept/drop)
GET  /acl/acl-filter[name=FILTER-IN][type=ipv4]/entry[sequence-id=10]/statistics  # match-packets, match-octets
GET  /acl/acl-filter[name=FILTER6-IN][type=ipv6]/entry[sequence-id=*]

--- S7: PING / TRACEROUTE ---
# CLI method only. Always specify network-instance.
# !!! WARNING: ALWAYS pass -c <N>. Omitting -c = unbounded ping = JSON-RPC connection hangs
# !!! indefinitely. Surfaces as transport timeout, NOT a command error. No exceptions. !!!
CLI  ping <dst> -c <N> network-instance <ni>
CLI  ping 10.0.0.1 -c 3 network-instance default
CLI  ping 8.8.8.8 -c 5 -s 1400 network-instance mgmt
CLI  ping 8.8.8.8 -c 5 -I 192.168.0.1 network-instance mgmt
# Flags: -c count(MANDATORY) -s packet-size -I source-ip
CLI  traceroute <dst> network-instance <ni>
CLI  tcptraceroute <dst> network-instance <ni>

CLI  show network-instance default protocols bgp neighbor X detail
GET  /interface[name=<uplink>]/oper-state
CLI  ping X -c 3 network-instance default

# ROUTING PROBLEM

CLI  show network-instance default route-table ipv4-unicast prefix X.X.X.X/YY

CLI  ping <dst> -c 3 network-instance default
